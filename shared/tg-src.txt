// server.ts: small API for the viewer's Solo mode (PLAN section 12).
//   GET  /api/health
//   GET  /api/rival?player=<id>   newest rival playbook for that player (or null)
//   POST /api/solo                save a finished solo battle (split "human")
import http from "node:http";
import path from "node:path";
import { fileURLToPath } from "node:url";
import dotenv from "dotenv";
import { MongoClient, ObjectId } from "mongodb";
import { validatePlaybook } from "./executor.js";
import { updateRival } from "./rival.js";

const here = path.dirname(fileURLToPath(import.meta.url));
dotenv.config({ path: path.resolve(here, "../../.env"), quiet: true });

const PORT = Number(process.env.API_PORT ?? 8787);
const DB_NAME = process.env.MONGODB_DB ?? "training_grounds";
const PLAYER_RE = /^[\w-]{1,32}$/;
const HEX24 = /^[0-9a-f]{24}$/i;

function send(res: http.ServerResponse, code: number, body: unknown) {
  res.writeHead(code, { "Content-Type": "application/json" });
  res.end(JSON.stringify(body));
}

function readJson(req: http.IncomingMessage): Promise<any> {
  return new Promise((resolve, reject) => {
    const chunks: Buffer[] = [];
    let size = 0;
    req.on("data", (c: Buffer) => {
      size += c.length;
      if (size > 2_000_000) { reject(new Error("body too large")); req.destroy(); }
      else chunks.push(c);
    });
    req.on("end", () => {
      try { resolve(JSON.parse(Buffer.concat(chunks).toString("utf8") || "{}")); }
      catch { reject(new Error("invalid JSON")); }
    });
    req.on("error", reject);
  });
}

async function main() {
  const uri = process.env.MONGODB_URI;
  if (!uri) { console.error("MONGODB_URI is empty. Put it in training-grounds/.env"); process.exit(1); }
  const client = new MongoClient(uri);
  await client.connect();
  const db = client.db(DB_NAME);

  const server = http.createServer(async (req, res) => {
    try {
      const url = new URL(req.url ?? "/", "http://localhost");

      if (req.method === "GET" && url.pathname === "/api/health") return send(res, 200, { ok: true });

      if (req.method === "GET" && url.pathname === "/api/rival") {
        const player = url.searchParams.get("player") ?? "";
        if (!PLAYER_RE.test(player)) return send(res, 400, { error: "bad player id" });
        const pb = await db.collection("playbooks")
          .find({ run_id: `rival-${player}`, side: "npc" })
          .sort({ generation: -1, created_at: -1 }).limit(1).next();
        if (!pb) return send(res, 200, { rival: null });
        return send(res, 200, { rival: {
          id: String(pb._id), generation: pb.generation ?? 0, persona: pb.persona ?? null,
          rationale: pb.rationale ?? "", playbook: validatePlaybook(pb.playbook).playbook,
        } });
      }

      if (req.method === "POST" && url.pathname === "/api/solo") {
        const body = await readJson(req);
        const player = String(body.player ?? "");
        if (!PLAYER_RE.test(player)) return send(res, 400, { error: "bad player id" });
        const r = body.result ?? {};
        if (r.winner !== "player" && r.winner !== "npc") return send(res, 400, { error: "bad result" });
        const runId = `rival-${player}`;
        const ins = await db.collection("battles").insertOne({
          run_id: runId,
          generation: Number.isInteger(body.rival_generation) ? body.rival_generation : 0,
          condition: "adaptive",
          split: "human",
          player,
          playbook_id: typeof body.rival_playbook_id === "string" && HEX24.test(body.rival_playbook_id)
            ? new ObjectId(body.rival_playbook_id) : null,
          party: Array.isArray(body.party) ? body.party : null,
          map_id: String(body.map_id ?? ""),
          npc_id: String(body.npc_id ?? ""),
          seed: Number(body.seed) || 0,
          winner: r.winner, reason: r.reason, turns: r.turns, score: r.score,
          summary: body.summary ?? null,
          events: Array.isArray(r.log) ? r.log : [],
          created_at: new Date(),
        });
        const n = await db.collection("battles").countDocuments({ run_id: runId, split: "human" });
        console.log(`saved solo battle for ${player}: ${r.winner === "player" ? "human won" : "rival won"} (${n} on record)`);
        const rival = process.env.RIVAL_RULES === "off" ? null : await updateRival(db, player);
        if (rival) console.log(`rival-${player} gen ${rival.generation}: ${rival.rationale}`);
        return send(res, 200, { ok: true, id: String(ins.insertedId), battles: n, rival });
      }

      send(res, 404, { error: "not found" });
    } catch (e) {
      send(res, 500, { error: (e as Error).message });
    }
  });

  server.listen(PORT, () => console.log(`API on http://localhost:${PORT} (db ${DB_NAME})`));
}

main();

  if (BEATS[attacker] === defender) return TRIANGLE_ADV;
  if (BEATS[defender] === attacker) return TRIANGLE_DIS;

  return null;

// ---------- Seeded RNG (mulberry32) ----------
export class Rng {
  private s: number;
  constructor(seed: number) { this.s = seed >>> 0; }
  next(): number {
    this.s = (this.s + 0x6d2b79f5) >>> 0;
    let t = this.s;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  }
  int(min: number, max: number): number { return min + Math.floor(this.next() * (max - min + 1)); }
  chance(pct: number): boolean { return this.next() * 100 < pct; }
  pick<T>(arr: readonly T[]): T { return arr[Math.floor(this.next() * arr.length)]; }
}

// ---------- Positions ----------
export interface Pos { x: number; y: number; }
export const inBounds = (p: Pos) => p.x >= 0 && p.y >= 0 && p.x < GRID_SIZE && p.y < GRID_SIZE;
export const dist = (a: Pos, b: Pos) => Math.abs(a.x - b.x) + Math.abs(a.y - b.y);
export const samePos = (a: Pos, b: Pos) => a.x === b.x && a.y === b.y;

// ---------- Units ----------
// UnitSpec mirrors the party JSON in shared/PLAN.md section 5.2
export interface UnitSpec {
  id: string;
  class: UnitClass;
  weapon: Weapon;
  attribute: Attribute;
  level: number;
  exp: number;
  star: number;
  promotion: number;          // 0 = base, 1-3 = tiers reached
  items?: Item[];             // PLAN section 11: max 2, restored every battle
  subskill: Subskill | null;
}

export interface Unit {
  id: string;
  team: Team;
  cls: UnitClass;
  weapon: Weapon;
  attribute: Attribute;
  level: number;
  subskill: Subskill | null;
  stats: Stats;               // effective stats (base + growth + subskill)
  maxHp: number;
  hp: number;
  pos: Pos;
  alive: boolean;
  burnTurns: number;
  mark: Attribute | null;     // attribute left by the last hit; a different one sets off a response
  items: Item[];
  expGained: number;          // earned this battle, read by Quartermaster
}

export function computeStats(cls: UnitClass, level: number, subskill: Subskill | null): Stats {
  const { base, growth } = CLASSES[cls];
  const lv = Math.max(0, level - 1);
  const s: Stats = {
    hp: base.hp + growth.hp * lv,
    atk: base.atk + growth.atk * lv,
    mag: base.mag + growth.mag * lv,
    def: base.def + growth.def * lv,
    spd: base.spd + growth.spd * lv,
  };
  if (subskill) {
    const b = SUBSKILL_BONUS[subskill];
    for (const k of Object.keys(b) as (keyof Stats)[]) s[k] += b[k] ?? 0;
  }
  return s;
}

export function makeUnit(spec: UnitSpec, team: Team, pos: Pos): Unit {
  const stats = computeStats(spec.class, spec.level, spec.subskill);
  return {
    id: spec.id, team, cls: spec.class, weapon: spec.weapon, attribute: spec.attribute,
    level: spec.level, subskill: spec.subskill, stats, maxHp: stats.hp, hp: stats.hp,
    pos: { ...pos }, alive: true, burnTurns: 0, mark: null, items: (spec.items ?? [Item.Vulnerary]).slice(0, MAX_ITEMS_PER_UNIT), expGained: 0,
  };
}

// ---------- Battle log events ----------
export type BattleEvent =
  | { t: "turn_start"; turn: number; team: Team }
  | { t: "move"; turn: number; unit: string; from: Pos; to: Pos }
  | { t: "attack"; turn: number; attacker: string; defender: string; hit: boolean;
      damage: number; triangle: number; reaction: Reaction | null; terrain: Terrain }
  | { t: "death"; turn: number; unit: string; killer: string | null; terrain: Terrain;
      triangleDisadvantage: boolean }
  | { t: "heal"; turn: number; unit: string; amount: number; source: "fort" | "item" }
  | { t: "burn"; turn: number; unit: string; damage: number }
  | { t: "evade"; turn: number; unit: string; attacker: string; terrain: Terrain }
  | { t: "end"; turn: number; winner: Team; reason: "rout" | "time" };

// ---------- Battle ----------
export interface Battle {
  seed: number;
  rng: Rng;
  mapId: string;
  grid: Terrain[][];          // grid[y][x]
  units: Unit[];
  turn: number;
  log: BattleEvent[];
  winner: Team | null;
}

export function terrainAt(b: Battle, p: Pos): Terrain { return b.grid[p.y][p.x]; }

export function unitAt(b: Battle, p: Pos): Unit | undefined {
  return b.units.find(u => u.alive && samePos(u.pos, p));
}

export function aliveOf(b: Battle, team: Team): Unit[] {
  return b.units.filter(u => u.alive && u.team === team);
}

export function hpRatio(b: Battle, team: Team): number {
  const all = b.units.filter(u => u.team === team);
  const max = all.reduce((s, u) => s + u.maxHp, 0);
  const cur = all.reduce((s, u) => s + (u.alive ? u.hp : 0), 0);
  return max === 0 ? 0 : cur / max;
}

export function getUnit(b: Battle, id: string): Unit {
  const u = b.units.find(u => u.id === id);
  if (!u) throw new Error(`unknown unit ${id}`);
  return u;
}

// rules.ts: combat math and turn-start effects, based on the Purifier SRPG design docs.
import {
  Attribute, BURN_DOT_PCT, BURN_TURNS, EXP_HIT, EXP_KILL, HIT_MAX, HIT_MIN,
  ITEM_HEAL_PCT, REACTIONS, Reaction, Subskill, TERRAIN, Team, UnitClass, WEAPON_RANGE,
  isMagic, triangleMod, WEAPON_POWER, ITEM_CLEARS_STATUS,
} from "./data.js";
import { Battle, Unit, aliveOf, dist, terrainAt } from "./state.js";

// ---------- Tunables ----------
export const DOUBLE_ATTACK_SPD_GAP = 4; // follow-up hit when SPD gap is at least this
export const EVADE_BASE = 10;           // design doc: evade = (SPD diff) + 10
export const EVADE_CAP = 60;            // keeps forest + high SPD from becoming untouchable
export const MIN_DAMAGE = 1;            // avoids 10-turn stalemates
export const RESPONSE_MAG_SCALE = 0.01; // response damage x (1 + MAG x 0.01)
export const PYROMANCER_MULT = 1.2;
export const BULWARK_MULT = 1.2;

// Mechanics doc: magic effectiveness by target type
export const MAGIC_EFFECTIVENESS: Record<UnitClass, number> = {
  [UnitClass.Infantry]: 1,
  [UnitClass.Armored]: 2,
  [UnitClass.Cavalier]: 1,
  [UnitClass.Winged]: 0.5,
  [UnitClass.Mage]: 1,
};

const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v));

// ---------- Elemental responses (symmetric pairs) ----------
export function pairReaction(a: Attribute, b: Attribute): Reaction | null {
  if (a === b) return null;
  const has = (x: Attribute) => a === x || b === x;
  if (has(Attribute.Fire) && has(Attribute.Electric)) return Reaction.Explosion;
  if (has(Attribute.Water) && has(Attribute.Electric)) return Reaction.ElectricShock;
  if (has(Attribute.Fire) && has(Attribute.Water)) return Reaction.CoolFire;
  return null;
}

// ---------- Hit / evade ----------
export function evadeChance(b: Battle, att: Unit, def: Unit): number {
  const forestBonus = TERRAIN[terrainAt(b, def.pos)].evadeFromSpd ? def.stats.spd : 0;
  return clamp(def.stats.spd - att.stats.spd + EVADE_BASE + forestBonus, 0, EVADE_CAP);
}

export function hitChance(b: Battle, att: Unit, def: Unit): number {
  return clamp(100 - evadeChance(b, att, def), HIT_MIN, HIT_MAX);
}

export function strikesFor(att: Unit, def: Unit): number {
  return att.stats.spd - def.stats.spd >= DOUBLE_ATTACK_SPD_GAP ? 2 : 1;
}

export function canCounter(def: Unit, att: Unit): boolean {
  if (!def.alive) return false;
  const [min, max] = WEAPON_RANGE[def.weapon];
  const d = dist(def.pos, att.pos);
  return d >= min && d <= max;
}

// ---------- Damage ----------
export interface DamagePreview {
  damage: number;
  triangle: number;
  reaction: Reaction | null;
}

// Pure: no RNG, no state changes. NPC AI and the executor use this to evaluate targets.
export function previewDamage(att: Unit, def: Unit): DamagePreview {
  const triangle = triangleMod(att.weapon, def.weapon);
  const raw = isMagic(att.weapon)
    ? ((att.stats.mag + att.stats.atk) / 2 + WEAPON_POWER[att.weapon]) * MAGIC_EFFECTIVENESS[def.cls]
    : (att.stats.atk + WEAPON_POWER[att.weapon]) * triangle;
  let dmg = Math.max(MIN_DAMAGE, raw - def.stats.def);

  const reaction = def.mark && def.mark !== att.attribute ? pairReaction(att.attribute, def.mark) : null;
  if (reaction) {
    let mult = REACTIONS[reaction].damageMult * (1 + att.stats.mag * RESPONSE_MAG_SCALE);
    if (att.subskill === Subskill.Pyromancer && att.attribute === Attribute.Fire) mult *= PYROMANCER_MULT;
    dmg *= mult;
  }
  return { damage: Math.max(MIN_DAMAGE, Math.round(dmg)), triangle, reaction };
}

export function expectedDamage(b: Battle, att: Unit, def: Unit): number {
  return previewDamage(att, def).damage * (hitChance(b, att, def) / 100) * strikesFor(att, def);
}

// ---------- Applying effects ----------
export function applyDamage(b: Battle, u: Unit, dmg: number, killer: string | null, triangleDisadvantage: boolean): boolean {
  u.hp = Math.max(0, u.hp - dmg);
  if (u.hp > 0) return false;
  u.alive = false;
  u.mark = null;
  u.burnTurns = 0;
  b.log.push({ t: "death", turn: b.turn, unit: u.id, killer, terrain: terrainAt(b, u.pos), triangleDisadvantage });
  return true;
}

// One strike. Returns true if the defender died.
export function strike(b: Battle, att: Unit, def: Unit): boolean {
  const terrain = terrainAt(b, def.pos);
  const preview = previewDamage(att, def);
  const hit = b.rng.chance(hitChance(b, att, def));

  if (!hit) {
    b.log.push({ t: "evade", turn: b.turn, unit: def.id, attacker: att.id, terrain });
    b.log.push({ t: "attack", turn: b.turn, attacker: att.id, defender: def.id, hit: false,
                 damage: 0, triangle: preview.triangle, reaction: null, terrain });
    return false;
  }

  if (preview.reaction) {
    def.mark = null;
    if (preview.reaction === Reaction.CoolFire) def.burnTurns = 0;
  } else {
    def.mark = att.attribute;
  }
  if (att.attribute === Attribute.Fire && TERRAIN[terrain].burnable && preview.reaction !== Reaction.CoolFire) {
    def.burnTurns = BURN_TURNS;
  }

  b.log.push({ t: "attack", turn: b.turn, attacker: att.id, defender: def.id, hit: true,
               damage: preview.damage, triangle: preview.triangle, reaction: preview.reaction, terrain });
  att.expGained += EXP_HIT;
  const killed = applyDamage(b, def, preview.damage, att.id, triangleMod(def.weapon, att.weapon) < 1);
  if (killed) att.expGained += EXP_KILL;
  return killed;
}

// Full exchange: attack, counter, then one follow-up from the faster side. Max 2 strikes per side.
export function resolveCombat(b: Battle, att: Unit, def: Unit): void {
  strike(b, att, def);
  if (att.alive && def.alive && canCounter(def, att)) strike(b, def, att);
  if (!att.alive || !def.alive) return;
  if (strikesFor(att, def) === 2) strike(b, att, def);
  else if (strikesFor(def, att) === 2 && canCounter(def, att)) strike(b, def, att);
}

export function useItem(b: Battle, u: Unit): void {
  const item = u.items.shift();
  if (!item) return;
  const amount = Math.min(u.maxHp - u.hp, Math.max(1, Math.round(u.maxHp * ITEM_HEAL_PCT[item])));
  u.hp += amount;
  if (ITEM_CLEARS_STATUS[item]) { u.burnTurns = 0; u.mark = null; }
  b.log.push({ t: "heal", turn: b.turn, unit: u.id, amount, source: "item" });
}

// Fort heal and burn ticks for one team, at the start of its phase.
export function startOfPhase(b: Battle, team: Team): void {
  for (const u of aliveOf(b, team)) {
    const info = TERRAIN[terrainAt(b, u.pos)];
    if (info.healPct > 0 && u.hp < u.maxHp) {
      const pct = info.healPct * (u.subskill === Subskill.Bulwark ? BULWARK_MULT : 1);
      const amount = Math.min(u.maxHp - u.hp, Math.max(1, Math.round(u.maxHp * pct)));
      u.hp += amount;
      b.log.push({ t: "heal", turn: b.turn, unit: u.id, amount, source: "fort" });
    }
    if (u.burnTurns > 0) {
      const dmg = Math.max(1, Math.round(u.maxHp * BURN_DOT_PCT));
      u.burnTurns -= 1;
      b.log.push({ t: "burn", turn: b.turn, unit: u.id, damage: dmg });
      applyDamage(b, u, dmg, null, false);
    }
  }
}

import { Playbook, validatePlaybook } from "../../game/src/executor.js";

const KEY = "tg.player";

export function playerId(): string {
  try { return localStorage.getItem(KEY) || "player1"; } catch { return "player1"; }
}

export function setPlayerId(name: string): boolean {
  if (!/^[\w-]{1,32}$/.test(name)) return false;
  try { localStorage.setItem(KEY, name); } catch { /* storage blocked: keep default */ }
  return true;
}

export async function fetchRival(player: string): Promise<Rival | null> {
  try {
    const r = await fetch(`/api/rival?player=${encodeURIComponent(player)}`);
    if (!r.ok) return null;
    const j = await r.json();
    if (!j.rival) return null;
    return { ...j.rival, playbook: validatePlaybook(j.rival.playbook).playbook };
  } catch {
    return null;
  }
}

export async function postSolo(payload: unknown): Promise<string> {
  try {
    const r = await fetch("/api/solo", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload),
    });
    const j = await r.json().catch(() => ({}));
    if (!r.ok) return `Not saved: ${j.error ?? r.status}`;
    const saved = `Saved · ${j.battles} battle${j.battles === 1 ? "" : "s"} on record.`;
    return j.rival ? `${saved} Rival Gen ${j.rival.generation} is ready: ${j.rival.rationale}` : `${saved} Your rival will study it.`;
  } catch {
    return "Not saved: API server isn't running (npm run server in game/).";
  }
}

function frameOf(b: Battle, event: BattleEvent | null, moved?: { id: string; to: Pos }): Frame {
  const units = Object.fromEntries(initialUnits(b).map(u =>
    [u.id, moved && u.id === moved.id ? { ...u, x: moved.to.x, y: moved.to.y } : u]));
  return { units, event, turn: b.turn, phase: null };
}

interface Forecast {
  dmg: number; hit: number; strikes: number; reaction: string | null; tri: number;
  counter: { dmg: number; hit: number; strikes: number } | null;
}

function forecast(b: Battle, u: Unit, to: Pos, def: Unit): Forecast {
  const orig = u.pos;
  u.pos = to;
  try {
    const pv = previewDamage(u, def);
    const f: Forecast = { dmg: pv.damage, hit: hitChance(b, u, def), strikes: strikesFor(u, def),
                          reaction: pv.reaction, tri: pv.triangle, counter: null };
    if (canCounter(def, u)) {
      const cp = previewDamage(def, u);
      f.counter = { dmg: cp.damage, hit: hitChance(b, def, u), strikes: strikesFor(def, u) };
    }
    return f;
  } finally {
    u.pos = orig;
  }
}

      const u = getUnit(b, queue[i]);
      if (!u.alive) { next(i + 1); return; }
      const a = npcPolicy.act(b, u);
      const s = b.log.length;
      if (a) {
        const err = applyAction(b, a);
        if (err) errors.push(`T${b.turn} npc: ${err}`);
      }
      animate(b.log.slice(s), () => next(i + 1));
    };
    animate(b.log.slice(start), () => next(0));
  };

  const act = (target: string | null, useItem: boolean) => {
    if (!sel || !dest) return;
    const unitId = sel;
    const start = b.log.length;
    const err = applyAction(b, { unit: unitId, to: dest, target, useItem });
    notice = err ?? "";
    if (err) errors.push(`T${b.turn} player: ${err}`);
    acted.add(unitId);
    sel = null;
    dest = null;
    mode = "busy";
    animate(b.log.slice(start), () => {
      if (checkEnd()) return;
      if (aliveOf(b, Team.Player).every(u => acted.has(u.id))) { enemyPhase(); return; }
      mode = "idle";
      lastEvent = null;
      render();
    });
  };

  // ---------- input ----------
  function confirm() {
    if (mode === "idle") {
      const u = unitAt(b, cursor);
      if (u && u.team === Team.Player && !acted.has(u.id)) { sel = u.id; mode = "move"; notice = ""; }
      else if (u && u.team === Team.Player) notice = `${u.id} has already acted.`;
      render();
      return;
    }
    if (mode === "move") {
      const u = getUnit(b, sel!);
      if (!reachableTiles(b, u).some(p => p.x === cursor.x && p.y === cursor.y)) {
        notice = "Can't move there.";
        render();
        return;
      }
      dest = { ...cursor };
      menu = [
        { key: "attack", label: "Attack", enabled: targetsFrom(b, u, dest).length > 0 },
        { key: "item", label: u.items.length ? `Item · ${cap(u.items[0])}` : "Item", enabled: u.items.length > 0 && u.hp < u.maxHp },
        { key: "wait", label: "Wait", enabled: true },
      ];
      menuIdx = menu.findIndex(m => m.enabled);
      notice = "";
      mode = "menu";
      render();
      return;
    }
    if (mode === "menu") {
      const m = menu[menuIdx];
      if (!m?.enabled) return;
      if (m.key === "attack") {
        targets = targetsFrom(b, getUnit(b, sel!), dest!);
        tIdx = 0;
        cursor = { ...targets[0].pos };
        mode = "target";
        render();
      } else act(null, m.key === "item");
      return;
    }
    if (mode === "target") act(targets[tIdx].id, false);
  }

  function cancel() {
    if (mode === "target") { mode = "menu"; cursor = { ...dest! }; }
    else if (mode === "menu") { mode = "move"; cursor = { ...dest! }; dest = null; }
    else if (mode === "move") { cursor = { ...getUnit(b, sel!).pos }; sel = null; mode = "idle"; }
    notice = "";
    render();
  }

  const arrow = (dx: number, dy: number) => {
    if (mode === "busy" || mode === "over") return;
    if (mode === "menu") {
      const step = dy || dx;
      do { menuIdx = (menuIdx + step + menu.length) % menu.length; } while (!menu[menuIdx].enabled);
      render();
      return;
    }
    if (mode === "target") {
      tIdx = (tIdx + (dx || dy) + targets.length) % targets.length;
      cursor = { ...targets[tIdx].pos };
      render();
      return;
    }
    cursor = { x: Math.max(0, Math.min(GRID_SIZE - 1, cursor.x + dx)), y: Math.max(0, Math.min(GRID_SIZE - 1, cursor.y + dy)) };
    render();
  };

  document.querySelectorAll<HTMLElement>("[data-dir]").forEach(btn => btn.addEventListener("click", () => {
    const [dx, dy] = btn.dataset.dir!.split(",").map(Number);
    arrow(dx, dy);
  }));
  byId("ok").onclick = () => { if (mode !== "busy" && mode !== "over") confirm(); };
  byId("confirm").onclick = () => { if (mode !== "busy" && mode !== "over") confirm(); };
  byId("cancel").onclick = () => { if (mode !== "busy" && mode !== "over") cancel(); };
  byId("endturn").onclick = () => { if (mode === "idle") enemyPhase(); };
  byId("rename").onclick = () => { const n = prompt("Player name (letters, numbers, - or _)", playerId()); if (n && setPlayerId(n)) go("solo"); };
  byId<HTMLInputElement>("speed").oninput = e => { S.speed = Number((e.target as HTMLInputElement).value); };
  canvas.addEventListener("click", ev => {
    if (mode === "busy" || mode === "over" || mode === "menu") return;
    const rect = canvas.getBoundingClientRect();
    const tile = {
      x: Math.min(GRID_SIZE - 1, Math.floor(((ev.clientX - rect.left) / rect.width) * GRID_SIZE)),
      y: Math.min(GRID_SIZE - 1, Math.floor(((ev.clientY - rect.top) / rect.height) * GRID_SIZE)),
    };
    if (mode === "target") {
      const i = targets.findIndex(t => t.pos.x === tile.x && t.pos.y === tile.y);
      if (i >= 0) { tIdx = i; cursor = tile; confirm(); }
      return;
    }
    cursor = tile;
    confirm();
  });
  const onKey = (e: KeyboardEvent) => {
    const tag = (e.target as HTMLElement).tagName;
    if (tag === "INPUT" || tag === "SELECT") return;
    const dirs: Record<string, [number, number]> = { ArrowUp: [0, -1], ArrowDown: [0, 1], ArrowLeft: [-1, 0], ArrowRight: [1, 0] };
    if (dirs[e.key]) { e.preventDefault(); arrow(...dirs[e.key]); }
    else if (e.key === "Enter" || e.key === " ") { e.preventDefault(); if (mode !== "busy" && mode !== "over") confirm(); }
    else if (e.key === "Escape" || e.key === "Backspace") { e.preventDefault(); if (mode !== "busy" && mode !== "over") cancel(); }
    else if (e.key === "e" || e.key === "E") { if (mode === "idle") enemyPhase(); }
  };
  window.addEventListener("keydown", onKey);
  setCleanup(() => {
    timers.forEach(t => window.clearTimeout(t));
    window.removeEventListener("keydown", onKey);
  });

  bindGo();
  render();
  fetchRival(playerId()).then(r => {
    rival = r;
    if (r) npcPolicy = playbookPolicy(r.playbook, "rival");
    playerPhase();
  });
}

// board.ts: canvas renderer for the battle map.
import { CLASSES, GRID_SIZE, Terrain } from "../../game/src/data.js";
import { Pos } from "../../game/src/state.js";
import { Frame, UnitView } from "./replay.js";
import { COLORS, TILE } from "./theme.js";

export interface DrawOptions {
  grid: Terrain[][];
  frame: Frame;
  cursor: Pos | null;
  blue: Pos[];
  red: Pos[];
}

export class BoardRenderer {
  readonly ctx: CanvasRenderingContext2D;
  readonly size = GRID_SIZE * TILE;
  private animation: number | null = null;
  private latest: DrawOptions | null = null;
  private readonly reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)");
  private readonly canvas: HTMLCanvasElement;

  constructor(canvas: HTMLCanvasElement) {
    this.canvas = canvas;
    const dpr = window.devicePixelRatio || 1;
    canvas.width = this.size * dpr;
    canvas.height = this.size * dpr;
    canvas.style.width = `${this.size}px`;
    canvas.style.height = `${this.size}px`;
    this.ctx = canvas.getContext("2d")!;
    this.ctx.scale(dpr, dpr);
  }

  draw(o: DrawOptions) {
    this.latest = o;
    if (this.animation === null && this.canvas.isConnected && !this.reducedMotion.matches) {
      this.animation = window.requestAnimationFrame(() => {
        this.animation = null;
        if (this.canvas.isConnected && this.latest) this.draw(this.latest);
      });
    }
    const c = this.ctx;
    c.clearRect(0, 0, this.size, this.size);
    this.terrain(o.grid);
    for (const p of o.blue) this.fillTile(p, COLORS.blueTile);
    for (const p of o.red) this.fillTile(p, COLORS.redTile);
    const e = o.frame.event;
    const us = o.frame.units;
    if (e?.t === "move") this.line(e.from, e.to, COLORS.team[us[e.unit].team], true);
    if (e?.t === "attack") this.line(us[e.attacker], us[e.defender], COLORS.cursor, false);
    for (const u of Object.values(us)) {
      const dying = e?.t === "death" && e.unit === u.id;
      if (u.alive || dying) this.unit(u, dying);
    }
    if (o.cursor) this.cursor(o.cursor);
    if (e?.t === "attack") {
      const d = us[e.defender];
      if (e.hit) {
        this.float(d, `-${e.damage}`, COLORS.damage);
        if (e.reaction) this.float(d, e.reaction.replace("_", " ").toUpperCase(), COLORS.reaction, 18);
      } else this.float(d, "MISS", COLORS.miss);
    }
    if (e?.t === "heal") this.float(us[e.unit], `+${e.amount}`, COLORS.heal);
    if (e?.t === "burn") this.float(us[e.unit], `-${e.damage} BURN`, COLORS.burn);
    if (e?.t === "death") this.float(us[e.unit], "DEFEATED", COLORS.damage);
  }

  private fillTile(p: Pos, color: string) {
    this.ctx.fillStyle = color;
    this.ctx.fillRect(p.x * TILE + 1, p.y * TILE + 1, TILE - 2, TILE - 2);
  }

  private terrain(grid: Terrain[][]) {
    const c = this.ctx;
    for (let y = 0; y < GRID_SIZE; y++) {
      for (let x = 0; x < GRID_SIZE; x++) {
        const t = grid[y][x];
        const px = x * TILE, py = y * TILE;
        const ground = c.createLinearGradient(px, py, px + TILE, py + TILE);
        ground.addColorStop(0, COLORS.terrain[t]);
        ground.addColorStop(1, t === Terrain.Water ? "#28527e" : t === Terrain.Forest ? "#285536" : t === Terrain.Fort ? "#67665f" : "#7e9b59");
        c.fillStyle = ground;
        c.fillRect(px, py, TILE, TILE);
        if (t === Terrain.Forest) {
          for (const [ox, oy] of [[0.24, 0.38], [0.7, 0.28], [0.52, 0.72]]) {
            const cx = px + ox * TILE, cy = py + oy * TILE;
            c.fillStyle = "rgba(10,25,17,0.28)";
            c.beginPath(); c.ellipse(cx + 3, cy + 10, 13, 6, 0, 0, Math.PI * 2); c.fill();
            c.fillStyle = "#614c32"; c.fillRect(cx - 2, cy, 4, 13);
            for (let layer = 0; layer < 3; layer++) {
              const top = cy - 18 + layer * 7, width = 8 + layer * 3;
              c.fillStyle = layer === 1 ? "#438653" : COLORS.forestDetail;
              c.beginPath(); c.moveTo(cx, top); c.lineTo(cx - width, top + 15); c.lineTo(cx + width, top + 15); c.closePath(); c.fill();
              c.strokeStyle = "rgba(166,206,119,0.32)"; c.lineWidth = 1;
              c.beginPath(); c.moveTo(cx, top + 2); c.lineTo(cx - width + 2, top + 13); c.stroke();
            }
          }
        } else if (t === Terrain.Fort) {
          c.fillStyle = "rgba(0,0,0,0.25)"; c.fillRect(px + 10, py + 15, TILE - 18, TILE - 23);
          c.fillStyle = "#b4afa0"; c.fillRect(px + 9, py + 15, TILE - 18, TILE - 26);
          c.strokeStyle = COLORS.fortDetail; c.lineWidth = 2;
          for (let row = 0; row < 4; row++) {
            const yy = py + 15 + row * 9;
            c.beginPath(); c.moveTo(px + 9, yy); c.lineTo(px + TILE - 9, yy); c.stroke();
            for (let col = 0; col < 3; col++) {
              const xx = px + 16 + col * 15 + (row % 2) * 7;
              c.beginPath(); c.moveTo(xx, yy); c.lineTo(xx, yy + 9); c.stroke();
            }
          }
          for (let i = 0; i < 4; i++) {
            const xx = px + 9 + i * ((TILE - 26) / 3);
            c.fillStyle = "#c8c0aa"; c.fillRect(xx, py + 8, 8, 10);
            c.fillStyle = COLORS.fortDetail; c.fillRect(xx + 6, py + 10, 2, 8);
          }
          c.fillStyle = "#41443f";
          c.beginPath(); c.arc(px + TILE / 2, py + 37, 7, Math.PI, 0); c.lineTo(px + TILE / 2 + 7, py + 53); c.lineTo(px + TILE / 2 - 7, py + 53); c.closePath(); c.fill();
        } else if (t === Terrain.Water) {
          for (let i = 0; i < 3; i++) {
            const yy = py + 14 + i * 17 + (this.reducedMotion.matches ? 0 : Math.sin(performance.now() / 1100 + x + y + i) * 1.2), shift = ((x + y + i) % 3) * 3;
            c.strokeStyle = i === 1 ? COLORS.waterDetail : "rgba(176,218,240,0.45)"; c.lineWidth = 2;
            c.beginPath(); c.moveTo(px + 6 + shift, yy);
            c.bezierCurveTo(px + 20, yy - 6, px + 36, yy + 6, px + TILE - 6 - shift, yy); c.stroke();
            c.strokeStyle = "rgba(210,239,255,0.55)"; c.lineWidth = 1;
            c.beginPath(); c.moveTo(px + 13 + shift, yy + 3); c.lineTo(px + 23 + shift, yy + 3); c.stroke();
          }
        } else {
          c.strokeStyle = "rgba(53,79,38,0.2)"; c.lineWidth = 1;
          for (const [ox, oy] of [[10, 15], [46, 44], [19, 53]]) {
            c.beginPath(); c.moveTo(px + ox - 2, py + oy); c.lineTo(px + ox, py + oy - 4); c.lineTo(px + ox + 2, py + oy); c.stroke();
          }
        }
      }
    }
    c.strokeStyle = COLORS.grid; c.lineWidth = 1;
    for (let i = 0; i <= GRID_SIZE; i++) {
      c.beginPath(); c.moveTo(i * TILE + 0.5, 0); c.lineTo(i * TILE + 0.5, this.size); c.stroke();
      c.beginPath(); c.moveTo(0, i * TILE + 0.5); c.lineTo(this.size, i * TILE + 0.5); c.stroke();
    }
  }

  private line(a: Pos, b: Pos, color: string, dashed: boolean) {
    const c = this.ctx;
    c.strokeStyle = color; c.lineWidth = 3;
    c.setLineDash(dashed ? [6, 5] : []);
    c.beginPath();
    c.moveTo(a.x * TILE + TILE / 2, a.y * TILE + TILE / 2);
    c.lineTo(b.x * TILE + TILE / 2, b.y * TILE + TILE / 2);
    c.stroke();
    c.setLineDash([]);
  }

  private unit(u: UnitView, faded: boolean) {
    const c = this.ctx;
    const cx = u.x * TILE + TILE / 2, cy = u.y * TILE + TILE / 2 - 4, r = TILE * 0.34;
    c.globalAlpha = faded ? 0.35 : 1;
    c.fillStyle = COLORS.team[u.team];
    c.beginPath(); c.arc(cx, cy, r, 0, Math.PI * 2); c.fill();
    c.strokeStyle = u.team === "player" ? COLORS.gold : "#fff"; c.lineWidth = 2; c.stroke();
    c.fillStyle = "#fff";
    c.font = `${Math.round(TILE * 0.42)}px "Apple Symbols", "Segoe UI Symbol", serif`;
    c.textAlign = "center"; c.textBaseline = "middle";
    c.fillText(CLASSES[u.cls].glyph + "\uFE0E", cx, cy + 1);
    c.fillStyle = COLORS.attr[u.attribute];
    c.beginPath(); c.arc(cx + r * 0.78, cy - r * 0.78, 5.5, 0, Math.PI * 2); c.fill();
    c.strokeStyle = "#111"; c.lineWidth = 1.5; c.stroke();
    const w = TILE * 0.72, h = 5, bx = cx - w / 2, by = u.y * TILE + TILE - 10;
    const ratio = u.hp / u.maxHp;
    c.fillStyle = COLORS.hpBack; c.fillRect(bx, by, w, h);
    c.fillStyle = ratio > 0.6 ? COLORS.hpGood : ratio > 0.3 ? COLORS.hpMid : COLORS.hpLow;
    c.fillRect(bx, by, w * ratio, h);
    c.globalAlpha = 1;
    if (faded) {
      c.save();
      c.strokeStyle = "#180e15"; c.lineWidth = 7;
      c.beginPath(); c.moveTo(cx - 12, cy - 12); c.lineTo(cx + 12, cy + 12);
      c.moveTo(cx + 12, cy - 12); c.lineTo(cx - 12, cy + 12); c.stroke();
      c.strokeStyle = COLORS.damage; c.lineWidth = 3; c.stroke();
      c.restore();
    }
  }

  private cursor(p: Pos) {
    const c = this.ctx;
    const x = p.x * TILE + 3, y = p.y * TILE + 3, s = TILE - 6, k = 14;
    c.save();
    const pulse = this.reducedMotion.matches ? 0.5 : (Math.sin(performance.now() / 350) + 1) / 2;
    c.shadowColor = COLORS.cursor; c.shadowBlur = 3 + pulse * 5;
    c.strokeStyle = "#220d14"; c.lineWidth = 6;
    c.beginPath();
    c.moveTo(x, y + k); c.lineTo(x, y); c.lineTo(x + k, y);
    c.moveTo(x + s - k, y); c.lineTo(x + s, y); c.lineTo(x + s, y + k);
    c.moveTo(x, y + s - k); c.lineTo(x, y + s); c.lineTo(x + k, y + s);
    c.moveTo(x + s - k, y + s); c.lineTo(x + s, y + s); c.lineTo(x + s, y + s - k);
    c.stroke();
    c.strokeStyle = COLORS.cursor; c.lineWidth = 3 + pulse; c.stroke();
    c.restore();
  }

  private float(u: UnitView, text: string, color: string, offset = 0) {
    const c = this.ctx;
    const size = /^[+-]\d+$/.test(text) ? 25 : 17;
    c.save();
    c.font = `bold ${size}px Inter, system-ui, sans-serif`;
    const half = c.measureText(text).width / 2 + 5;
    const x = Math.max(half, Math.min(this.size - half, u.x * TILE + TILE / 2));
    // Keep reactions above damage while retaining labels on the top row.
    const y = Math.max(size / 2 + 4, u.y * TILE + (offset ? 12 : 32));
    c.textAlign = "center"; c.textBaseline = "middle";
    c.lineJoin = "round";
    c.lineWidth = 5; c.strokeStyle = "rgba(0,0,0,0.95)"; c.strokeText(text, x, y);
    c.fillStyle = color; c.fillText(text, x, y);
    c.restore();
  }
}

function drawSpark(canvas: HTMLCanvasElement, full: GenExport[], frozen: GenExport[], sel: number) {
  const W = Math.max(320, (canvas.parentElement?.clientWidth ?? 600) - 30), H = 200, dpr = window.devicePixelRatio || 1;
  canvas.width = W * dpr; canvas.height = H * dpr;
  canvas.style.width = `${W}px`; canvas.style.height = `${H}px`;
  const c = canvas.getContext("2d")!;
  c.scale(dpr, dpr);
  const pad = 34, n = Math.max(full.length, frozen.length, 2);
  const X = (i: number) => pad + (i * (W - 2 * pad)) / (n - 1);
  const Y = (v: number) => H - pad - v * (H - 2 * pad);
  c.strokeStyle = "rgba(201,162,74,0.18)"; c.lineWidth = 1;
  c.font = "13px Inter, sans-serif"; c.fillStyle = "#98a1b6";
  for (const v of [0, 0.5, 1]) {
    c.beginPath(); c.moveTo(pad, Y(v)); c.lineTo(W - pad, Y(v)); c.stroke();
    c.fillText(`${v * 100}%`, 0, Y(v) - 3);
  }
  const series = (gs: GenExport[], color: string, dash: number[]) => {
    const pts = gs.filter(g => g.win_rate != null);
    if (!pts.length) return;
    c.strokeStyle = color; c.lineWidth = 3; c.setLineDash(dash);
    c.beginPath();
    pts.forEach((g, i) => (i ? c.lineTo(X(g.generation), Y(g.win_rate!)) : c.moveTo(X(g.generation), Y(g.win_rate!))));
    c.stroke(); c.setLineDash([]);
    c.fillStyle = color;
    for (const g of pts) { c.beginPath(); c.arc(X(g.generation), Y(g.win_rate!), 4, 0, Math.PI * 2); c.fill(); }
  };
  series(frozen, "#7d8699", [4, 4]);
  series(full, "#ecd28c", []);
  const s = full[sel];
  if (s?.win_rate != null) {
    c.strokeStyle = "#fff"; c.lineWidth = 2;
    c.beginPath(); c.arc(X(s.generation), Y(s.win_rate), 5, 0, Math.PI * 2); c.stroke();
  }
}

function battleScreen() {

            <button class="btn sm" data-dir="-1,0">◀</button><button class="btn sm sel" id="sel" title="Select (Enter)">SEL</button><button class="btn sm" data-dir="1,0">▶</button>

              <button class="btn sm" id="last" title="Last">⏭</button>
            </div>
            <label class="field inline">Speed <input type="range" id="speed" min="1" max="10" value="${S.speed}" /></label>
            <div class="hint">Arrows: cursor · Enter: select · Space: play · Z/X: step</div>
          </div>
        </div>
      </div>
      <div class="side-col">
        <div class="panel"><h3>Coach · ${esc(g.best?.persona ?? "playbook")}</h3>
          <p class="quote small">${esc(g.best?.rationale || "No rationale recorded.")}</p>
          <div class="tags">
            <span class="chip">${cap(pb.formation)}</span>
            <span class="chip">Advance T${pb.advance_turn}</span>
            <span class="chip">${pb.target_priority.map(cap).join(" → ")}</span>
            ${pb.focus_fire ? `<span class="chip">Focus fire</span>` : ""}
            ${pb.use_fire_on_forest ? `<span class="chip">Fire on forest</span>` : ""}

        <div class="panel"><h3>Battle Log</h3><ol class="log" id="log"></ol></div>

    </div>

    return `<li data-i="${i}" class="${e.t} ${side ? `side-${side}` : ""}">${esc(describe(e))}</li>`;

    const f = frames[i];
    let unitId: string | null = null;
    let from: Frame = f;
    if (selected && f.units[selected]?.alive) unitId = selected;
    else if (f.event?.t === "move" && i > 0) { unitId = f.event.unit; from = frames[i - 1]; }
    const blue: Pos[] = [];
    const red: Pos[] = [];
    if (unitId) {
      const b = battleAt(base, from);
      const u = getUnit(b, unitId);
      blue.push(...reachableTiles(b, u));
      const mv = CLASSES[u.cls].move;
      for (let y = 0; y < GRID_SIZE; y++) for (let x = 0; x < GRID_SIZE; x++) {
        const d = dist({ x, y }, u.pos);
        if (d > 0 && d <= mv && !blue.some(p => p.x === x && p.y === y)) red.push({ x, y });
      }
    }
    return { blue, red };
  };

  const unitPanel = (f: Frame) => {
    const t = base.grid[cursor.y][cursor.x];
    const tile = `<div class="tile-info"><span class="chip" style="color:${COLORS.terrain[t]}">${cap(t)}</span> ${TERRAIN_NOTES[t]}</div>`;
    const v = Object.values(f.units).find(u => u.alive && u.x === cursor.x && u.y === cursor.y)
      ?? (selected ? f.units[selected] : undefined);
    if (!v) {
      $("unitpanel").innerHTML = `<h3>Unit</h3><p class="muted">Move the cursor onto a unit to inspect it. Select shows its movement range.</p>${tile}`;
      return;
    }
    const u = base.units.find(x => x.id === v.id)!;
    const info = CLASSES[u.cls];
    const r = v.hp / v.maxHp;
    $("unitpanel").innerHTML = `<h3>${v.team === "player" ? "Your Unit" : "Enemy Unit"}</h3>
      <div class="uhead"><div class="uglyph" style="background:${COLORS.team[v.team]}">${info.glyph}\uFE0E</div>

      <div class="muted small">HP ${v.hp} / ${v.maxHp}${v.alive ? "" : " · defeated"}</div>
      <div class="stats">
        <span>ATK</span><span>${u.stats.atk}</span><span>MAG</span><span>${u.stats.mag}</span>

      ${tile}`;
  };

  const render = () => {
    const f = frames[idx];
    const { blue, red } = overlays(idx);
    board.draw({ grid: base.grid, frame: f, cursor, blue, red });
    $("turninfo").textContent = `Turn ${f.turn}${f.phase ? ` · ${f.phase === "player" ? "Player" : "Enemy"} Phase` : ""} · ${idx}/${frames.length - 1}`;
    const last = idx === frames.length - 1;
    const banner = $("banner");
    banner.className = `banner ${last ? (result.winner === "player" ? "win" : "loss") : ""}`;
    banner.textContent = last ? (result.winner === "player" ? "VICTORY" : "DEFEAT") : "";
    unitPanel(f);
    $("log").querySelector(".current")?.classList.remove("current");
    const li = $("log").querySelector(`[data-i="${idx}"]`);
    if (li) { li.classList.add("current"); li.scrollIntoView({ block: "nearest" }); }
  };

  const follow = () => {
    const f = frames[idx];
    const e = f.event;
    if (!e) return;
    if (e.t === "move") cursor = { ...e.to };
    else if (e.t === "attack") cursor = { x: f.units[e.attacker].x, y: f.units[e.attacker].y };
    else if ("unit" in e && f.units[e.unit]) cursor = { x: f.units[e.unit].x, y: f.units[e.unit].y };
  };

  const step = (i: number) => { idx = Math.max(0, Math.min(frames.length - 1, i)); follow(); render(); };
  const stop = () => {
    if (timer !== null) window.clearInterval(timer);
    timer = null;
    $("play").textContent = "Play";
  };
  const play = () => {
    if (timer !== null) return;
    selected = null;
    if (idx >= frames.length - 1) step(0);
    $("play").textContent = "Pause";
    timer = window.setInterval(() => (idx >= frames.length - 1 ? stop() : step(idx + 1)), 1150 - S.speed * 105);

    stop();
    cursor = { x: Math.max(0, Math.min(GRID_SIZE - 1, cursor.x + dx)), y: Math.max(0, Math.min(GRID_SIZE - 1, cursor.y + dy)) };
    render();
  };
  const select = () => {
    stop();
    const v = Object.values(frames[idx].units).find(u => u.alive && u.x === cursor.x && u.y === cursor.y);
    selected = v && selected !== v.id ? v.id : null;
    render();
  };

    const [dx, dy] = b.dataset.dir!.split(",").map(Number);
    moveCursor(dx, dy);
  }));

  $("first").onclick = () => { stop(); step(0); };
  $("prev").onclick = () => { stop(); step(idx - 1); };
  $("next").onclick = () => { stop(); step(idx + 1); };
  $("last").onclick = () => { stop(); step(frames.length - 1); };
  $("play").onclick = () => (timer === null ? play() : stop());

    S.speed = Number((e.target as HTMLInputElement).value);

  };

  $("log").querySelectorAll<HTMLElement>("li").forEach(li =>
    li.addEventListener("click", () => { stop(); step(Number(li.dataset.i)); }));
  canvas.addEventListener("click", ev => {
    const rect = canvas.getBoundingClientRect();
    cursor = {
      x: Math.min(GRID_SIZE - 1, Math.floor(((ev.clientX - rect.left) / rect.width) * GRID_SIZE)),
      y: Math.min(GRID_SIZE - 1, Math.floor(((ev.clientY - rect.top) / rect.height) * GRID_SIZE)),
    };
    select();
  });
  const onKey = (e: KeyboardEvent) => {
    const tag = (e.target as HTMLElement).tagName;
    if (tag === "INPUT" || tag === "SELECT") return;
    const dirs: Record<string, [number, number]> = { ArrowUp: [0, -1], ArrowDown: [0, 1], ArrowLeft: [-1, 0], ArrowRight: [1, 0] };
    if (dirs[e.key]) { e.preventDefault(); moveCursor(...dirs[e.key]); }
    else if (e.key === "Enter") select();
    else if (e.key === " ") { e.preventDefault(); timer === null ? play() : stop(); }
    else if (e.key === "z" || e.key === "Z") { stop(); step(idx - 1); }
    else if (e.key === "x" || e.key === "X") { stop(); step(idx + 1); }
    else if (e.key === "Escape") go("menu");
  };
  window.addEventListener("keydown", onKey);
  cleanup = () => { stop(); window.removeEventListener("keydown", onKey); };

  bindGo();
  render();
  void TERRAIN;
  void TILE;
}
